// Oyuncular ve cevaplar ayrı tablolarda: aynı anda katılan ya da yazan kimse birbirini ezmez.

const HARFLER = 'ABCDEFGHJKLMNPRSTUVYZ';
const kodUret = () => Array.from({length: 4}, () => HARFLER[Math.floor(Math.random() * HARFLER.length)]).join('');

const BASLIK = {
  'content-type': 'application/json',
  'access-control-allow-origin': '*',
  'access-control-allow-headers': 'content-type',
  'access-control-allow-methods': 'GET,POST,OPTIONS',
};
const json = (v, s = 200) => new Response(JSON.stringify(v), {status: s, headers: BASLIK});
const karistir = a => a.map(v => [Math.random(), v]).sort((x, y) => x[0] - y[0]).map(p => p[1]);
const kirp = (v, n) => String(v == null ? '' : v).slice(0, n);

async function hazirla(env) {
  await env.DB.batch([
    env.DB.prepare('CREATE TABLE IF NOT EXISTS odalar (kod TEXT PRIMARY KEY, durum TEXT NOT NULL, guncel INTEGER NOT NULL)'),
    env.DB.prepare('CREATE TABLE IF NOT EXISTS oyuncular (kod TEXT NOT NULL, id TEXT PRIMARY KEY, ad TEXT NOT NULL, sira INTEGER NOT NULL, puan INTEGER NOT NULL DEFAULT 0, gizli TEXT)'),
    env.DB.prepare('CREATE TABLE IF NOT EXISTS cevaplar (kod TEXT NOT NULL, tur INTEGER NOT NULL, id TEXT NOT NULL, metin TEXT NOT NULL, sira INTEGER NOT NULL, PRIMARY KEY (kod, tur, id))'),
  ]);
  try { await env.DB.prepare('ALTER TABLE oyuncular ADD COLUMN hak INTEGER NOT NULL DEFAULT 1').run(); }
  catch (e) { /* sütun zaten var */ }
  try { await env.DB.prepare('ALTER TABLE oyuncular ADD COLUMN oy TEXT').run(); }
  catch (e) { /* sütun zaten var */ }
}

async function oylariSil(env, kod) {
  await env.DB.prepare('UPDATE oyuncular SET oy = NULL WHERE kod = ?').bind(kod).run();
}
// Aynı anda gelen son oylar fazı iki kere ilerletmesin: yalnız beklenen faz/tur duruyorsa yazar.
async function odaYazKosullu(env, kod, d, faz, tur) {
  const s = JSON.stringify(d), t = Date.now();
  const r = await env.DB.prepare(
    "UPDATE odalar SET durum = ?, guncel = ? WHERE kod = ? " +
    "AND json_extract(durum, '$.oyun.faz') = ? AND json_extract(durum, '$.oyun.tur') = ?"
  ).bind(s, t, kod, faz, tur).run();
  return !!(r.meta && r.meta.changes);
}

const bosDurum = () => ({acik: false, cevapAcik: false, soru: null, tur: 1, gercekYer: null, karisik: []});

async function odaOku(env, kod) {
  const r = await env.DB.prepare('SELECT durum FROM odalar WHERE kod = ?').bind(kod).first();
  return r ? JSON.parse(r.durum) : null;
}
async function odaYaz(env, kod, d) {
  const s = JSON.stringify(d), t = Date.now();
  await env.DB.prepare(
    'INSERT INTO odalar (kod, durum, guncel) VALUES (?, ?, ?) ON CONFLICT(kod) DO UPDATE SET durum = ?, guncel = ?'
  ).bind(kod, s, t, s, t).run();
}
async function oyuncuListe(env, kod) {
  const r = await env.DB.prepare(
    'SELECT id, ad, puan, gizli, hak, oy FROM oyuncular WHERE kod = ? ORDER BY sira ASC'
  ).bind(kod).all();
  return r.results || [];
}
async function cevapListe(env, kod, tur) {
  const r = await env.DB.prepare(
    'SELECT id, metin FROM cevaplar WHERE kod = ? AND tur = ? ORDER BY sira ASC'
  ).bind(kod, tur).all();
  return r.results || [];
}

async function gorunum(env, kod, d, id) {
  const oy = await oyuncuListe(env, kod);
  const cv = await cevapListe(env, kod, d.tur);
  const ben = oy.find(o => o.id === id) || null;
  const adlar = new Map(oy.map(o => [o.id, o.ad]));
  const yazdi = new Map(cv.map(c => [c.id, c.metin]));

  let cevaplar = [];
  if (d.acik) {
    const duzen = (d.karisik || []).filter(x => yazdi.has(x));
    for (const c of cv) if (!duzen.includes(c.id)) duzen.push(c.id);
    cevaplar = duzen.map(x => ({
      ad: d.soru && d.soru.anonim ? '' : (adlar.get(x) || ''),
      metin: yazdi.get(x),
    }));
    if (d.soru && d.soru.ekle && d.soru.cevap) {
      const yer = d.gercekYer == null ? cevaplar.length : Math.min(d.gercekYer, cevaplar.length);
      cevaplar.splice(yer, 0, {ad: '', metin: d.soru.cevap, gercek: !!d.cevapAcik});
    }
  }
  return {
    acik: !!d.acik,
    dogruAcik: !!d.cevapAcik,
    kisi: oy.length,
    yazan: cv.length,
    oyuncular: oy.map(o => ({ad: o.ad, puan: o.puan || 0})),
    sahip: !!ben && oy.length > 0 && oy[0].id === id,
    yazdim: !!ben && yazdi.has(ben.id),
    soru: d.soru ? d.soru.metin : null,
    cevap: d.cevapAcik && d.soru ? (d.soru.cevap || null) : null,
    gizli: ben ? ben.gizli || null : null,
    hak: ben ? (ben.hak == null ? 1 : ben.hak) : 0,
    tur: d.tur,
    cevaplar,
  };
}


/* ---------- Köylü ve vampir ---------- */
const GUNDUZ_SN = 180;

function vampirSayisi(n) { return n >= 11 ? 3 : (n >= 7 ? 2 : 1); }

function oylariTopla(oy) {
  const m = {};
  oy.forEach(x => { if (x.oy) m[x.id] = x.oy; });
  return m;
}

function vampirGorunum(d, oy, id) {
  const o = d.oyun;
  const adi = new Map(oy.map(x => [x.id, x.ad]));
  const oylar = oylariTopla(oy);
  const yasayan = o.yasayan.filter(x => adi.has(x));
  const benVampir = o.roller[id] === 'vampir';
  const benYasiyor = yasayan.includes(id);
  const oyVerenler = Object.keys(oylar);
  const gerekli = o.faz === 'gece'
    ? yasayan.filter(x => o.roller[x] === 'vampir')
    : yasayan;
  return {
    oyunVar: true,
    faz: o.faz,
    tur: o.tur,
    sonuc: o.sonuc || null,
    rol: o.roller[id] || null,
    yasiyorum: benYasiyor,
    arkadaslar: benVampir
      ? yasayan.filter(x => x !== id && o.roller[x] === 'vampir').map(x => adi.get(x))
      : [],
    yasayan: yasayan.map(x => ({id: x, ad: adi.get(x), oyVerdi: oyVerenler.includes(x)})),
    olenler: (o.olum || []).map(x => ({ad: adi.get(x.id) || '', faz: x.faz, rol: x.rol})),
    oyum: oylar[id] || null,
    oySayisi: oyVerenler.length,
    gereken: gerekli.length,
    kalanSn: o.faz === 'gunduz' && o.bitis ? Math.max(0, Math.round((o.bitis - Date.now()) / 1000)) : null,
    mesaj: o.mesaj || '',
    sahip: oy.length > 0 && oy[0].id === id,
  };
}

function cozumle(d, oy) {
  const o = d.oyun;
  const adi = new Map(oy.map(x => [x.id, x.ad]));
  const yasayan = o.yasayan.filter(x => adi.has(x));
  const oylar = oylariTopla(oy);
  let olen = null;

  if (o.faz === 'gece') {
    const vamp = yasayan.filter(x => o.roller[x] === 'vampir');
    const verenler = vamp.filter(x => oylar[x]);
    if (!verenler.length) return false;
    const secen = verenler[Math.floor(Math.random() * verenler.length)];
    olen = oylar[secen];
    o.mesaj = olen ? (adi.get(olen) || 'Biri') + ' gece öldürüldü.' : 'Gece kimse ölmedi.';
  } else {
    const sayim = {};
    Object.values(oylar).forEach(h => { if (h) sayim[h] = (sayim[h] || 0) + 1; });
    const en = Math.max(0, ...Object.values(sayim));
    const basta = Object.keys(sayim).filter(k => sayim[k] === en);
    if (en === 0 || basta.length !== 1) {
      o.mesaj = 'Oylar eşit, kimse idam edilmedi.';
    } else {
      olen = basta[0];
      o.mesaj = (adi.get(olen) || 'Biri') + ' idam edildi. Rolü: ' +
                (o.roller[olen] === 'vampir' ? 'vampir' : 'köylü') + '.';
    }
  }

  if (olen) {
    o.yasayan = o.yasayan.filter(x => x !== olen);
    o.olum.push({id: olen, faz: o.faz, rol: o.roller[olen], tur: o.tur});
  }

  const kalan = o.yasayan.filter(x => adi.has(x));
  const kalanVamp = kalan.filter(x => o.roller[x] === 'vampir').length;
  const kalanKoy = kalan.length - kalanVamp;
  if (kalanVamp === 0) { o.faz = 'bitti'; o.sonuc = 'koylu'; o.mesaj += ' Köylüler kazandı.'; }
  else if (kalanVamp >= kalanKoy) { o.faz = 'bitti'; o.sonuc = 'vampir'; o.mesaj += ' Vampirler kazandı.'; }
  else if (o.faz === 'gece') { o.faz = 'gunduz'; o.bitis = Date.now() + GUNDUZ_SN * 1000; }
  else { o.faz = 'gece'; o.tur += 1; o.bitis = null; }
  return true;
}

export default {
  async fetch(istek, env) {
    const url = new URL(istek.url);
    const y = url.pathname;

    if (istek.method === 'OPTIONS') return new Response(null, {status: 204, headers: BASLIK});
    if (!y.startsWith('/api/')) {
      return new Response(
        '<!doctype html><meta charset="utf-8"><title>Circle</title>' +
        '<meta name="viewport" content="width=device-width,initial-scale=1">' +
        '<body style="background:#111A24;color:#F6F1E4;font:16px/1.6 system-ui;padding:40px 22px;max-width:540px;margin:0 auto">' +
        '<p style="font-size:11px;letter-spacing:.2em;color:#1C8B9C;font-weight:600">CIRCLE</p>' +
        '<h1 style="font-size:30px;margin-top:10px">Oda sunucusu</h1>' +
        '<p style="opacity:.7;margin-top:14px">Odalar oyun sayfalarının içinden açılır. ' +
        'Kartın üzerindeki kodu okutun.</p>' +
        '<p style="margin-top:20px"><a style="color:#1C8B9C" href="https://oguber90-commits.github.io/circle/">Oyun sayfaları</a></p>',
        {headers: {'content-type': 'text/html; charset=utf-8'}});
    }

    try {
      await hazirla(env);
      const p = y.replace('/api/', '').split('/');
      const g = istek.method === 'POST' ? await istek.json().catch(() => ({})) : {};

      if (p[0] === 'oda' && p.length === 1 && istek.method === 'POST') {
        let kod;
        for (let i = 0; i < 8; i++) { kod = kodUret(); if (!(await odaOku(env, kod))) break; }
        const id = crypto.randomUUID();
        await odaYaz(env, kod, bosDurum());
        await env.DB.prepare('INSERT INTO oyuncular (kod, id, ad, sira, puan) VALUES (?, ?, ?, ?, 0)')
          .bind(kod, id, kirp(g.ad || 'Biri', 18), Date.now()).run();
        return json({kod, id});
      }

      const kod = (p[1] || '').toUpperCase();
      let d = await odaOku(env, kod);
      if (!d) return json({hata: 'oda yok'}, 404);

      if (p.length === 2 && istek.method === 'GET') {
        return json(await gorunum(env, kod, d, url.searchParams.get('id') || ''));
      }
      if (istek.method !== 'POST') return json({hata: 'yok'}, 405);

      if (p[2] === 'katil') {
        const ad = kirp(g.ad || 'Biri', 18);
        const v = await env.DB.prepare('SELECT id FROM oyuncular WHERE kod = ? AND lower(ad) = lower(?)')
          .bind(kod, ad).first();
        if (v) return json({id: v.id});
        const id = crypto.randomUUID();
        await env.DB.prepare('INSERT INTO oyuncular (kod, id, ad, sira, puan) VALUES (?, ?, ?, ?, 0)')
          .bind(kod, id, ad, Date.now()).run();
        return json({id});
      }

      const ben = await env.DB.prepare('SELECT id, ad FROM oyuncular WHERE id = ? AND kod = ?')
        .bind(kirp(g.id, 64), kod).first();
      if (!ben) return json({hata: 'odada degilsin'}, 403);

      if (p[2] === 'cevap') {
        await env.DB.prepare(
          'INSERT INTO cevaplar (kod, tur, id, metin, sira) VALUES (?, ?, ?, ?, ?) ' +
          'ON CONFLICT(kod, tur, id) DO UPDATE SET metin = ?'
        ).bind(kod, d.tur, ben.id, kirp(g.metin, 500), Date.now(), kirp(g.metin, 500)).run();

      } else if (p[2] === 'soru' || p[2] === 'tur' || p[2] === 'kapat' || p[2] === 'dagit') {
        const sonraki = d.tur + 1;
        if (p[2] === 'soru') {
          d.soru = {metin: kirp(g.metin, 600), cevap: kirp(g.cevap, 600),
                    anonim: !!g.anonim, ekle: !!g.ekle};
        } else if (p[2] === 'dagit') {
          const havuz = karistir((Array.isArray(g.gorevler) ? g.gorevler : []).slice());
          const oy = karistir((await oyuncuListe(env, kod)).map(o => o.id));
          const islem = [env.DB.prepare('UPDATE oyuncular SET gizli = NULL, hak = 1 WHERE kod = ?').bind(kod)];
          if (g.mod === 'rol') {
            // oyuncu sayısına göre az rol, kalanı çok rol
            const esik = Number(g.esik) || 8;
            const azSayi = oy.length >= esik ? 2 : 1;
            oy.forEach((id, i) => islem.push(
              env.DB.prepare('UPDATE oyuncular SET gizli = ? WHERE id = ?')
                .bind(kirp(i < azSayi ? g.rolAz : g.rolCok, 300), id)));
          } else if (g.mod === 'kod') {
            // iki kişiye anahtar, kalanlara tahminci metni
            const anahtar = kirp(havuz[0] || '', 400);
            oy.forEach((id, i) => islem.push(
              env.DB.prepare('UPDATE oyuncular SET gizli = ? WHERE id = ?')
                .bind(i < 2 ? (i === 0 ? 'KIRMIZI\n' : 'MAVI\n') + anahtar
                            : kirp(g.tekMetin || 'Sen tahmincisin', 300), id)));
          } else if (g.mod === 'yabanci') {
            // herkese aynı metin, rastgele bir kişiye farklı olan
            const ortak = kirp(havuz[0] || '', 300);
            const tek = kirp(g.tekMetin || 'Sen yabancısın', 300);
            oy.forEach((id, i) => islem.push(
              env.DB.prepare('UPDATE oyuncular SET gizli = ? WHERE id = ?').bind(i === 0 ? tek : ortak, id)));
          } else {
            // her oyuncuya kisiBasi kadar görev, hiçbiri kimseyle aynı değil
            const n = Math.max(1, Math.min(5, Number(g.kisiBasi) || 1));
            let sira = 0;
            for (const id of oy) {
              const pay = [];
              for (let j = 0; j < n && havuz.length; j++) pay.push(havuz[sira++ % havuz.length]);
              islem.push(env.DB.prepare('UPDATE oyuncular SET gizli = ? WHERE id = ?')
                .bind(kirp(pay.join('\n'), 900), id));
            }
          }
          await env.DB.batch(islem);
          d.soru = {metin: kirp(g.metin, 600), cevap: '', anonim: false, ekle: false};
        } else {
          d.soru = p[2] === 'kapat' ? null : d.soru;
          if (p[2] === 'kapat') await env.DB.prepare('UPDATE oyuncular SET gizli = NULL WHERE kod = ?').bind(kod).run();
        }
        d.tur = sonraki; d.acik = false; d.cevapAcik = false; d.karisik = []; d.gercekYer = null;
        await odaYaz(env, kod, d);

      } else if (p[2] === 'ac' || p[2] === 'cevapac') {
        const cv = await cevapListe(env, kod, d.tur);
        if (!(d.karisik || []).length) d.karisik = karistir(cv.map(c => c.id));
        if (d.gercekYer == null && d.soru && d.soru.ekle) {
          d.gercekYer = Math.floor(Math.random() * (d.karisik.length + 1));
        }
        d.acik = true;
        if (p[2] === 'cevapac') d.cevapAcik = true;
        await odaYaz(env, kod, d);

      } else if (p[2] === 'degistir') {
        const oy = await oyuncuListe(env, kod);
        const sen = oy.find(o => o.id === ben.id);
        if (!sen || (sen.hak == null ? 1 : sen.hak) < 1 || !sen.gizli) return json({hata: 'hak yok'}, 403);
        const satir = sen.gizli.split('\n');
        const ix = Math.max(0, Math.min(satir.length - 1, Number(g.sira) || 0));
        const kullanilan = new Set();
        oy.forEach(o => (o.gizli || '').split('\n').forEach(x => x && kullanilan.add(x)));
        const bos = (Array.isArray(g.havuz) ? g.havuz : []).filter(x => !kullanilan.has(x));
        if (!bos.length) return json({hata: 'yedek gorev kalmadi'}, 409);
        satir[ix] = kirp(bos[Math.floor(Math.random() * bos.length)], 300);
        await env.DB.prepare('UPDATE oyuncular SET gizli = ?, hak = 0 WHERE id = ?')
          .bind(satir.join('\n'), ben.id).run();

      } else if (p[2] === 'vampir') {
        const oy = await oyuncuListe(env, kod);
        const sira = karistir(oy.map(o => o.id));
        const vs = vampirSayisi(sira.length);
        const roller = {};
        sira.forEach((x, i) => roller[x] = i < vs ? 'vampir' : 'koylu');
        d.oyun = {faz: 'gece', tur: 1, yasayan: oy.map(o => o.id), roller,
                  olum: [], bitis: null, sonuc: null,
                  mesaj: 'Gece. Vampirler kurbanını seçiyor.'};
        d.soru = null; d.acik = false; d.cevapAcik = false;
        await oylariSil(env, kod);
        await odaYaz(env, kod, d);
        oy.forEach(x => x.oy = null);
        return json(vampirGorunum(d, oy, ben.id));

      } else if (p[2] === 'voy') {
        if (!d.oyun || d.oyun.faz === 'bitti') return json({hata: 'oyun yok'}, 409);
        const o = d.oyun;
        const once = await oyuncuListe(env, kod);
        const adi = new Set(once.map(x => x.id));
        const yasayan = o.yasayan.filter(x => adi.has(x));
        const yetkili = o.faz === 'gece' ? (o.roller[ben.id] === 'vampir') : true;
        if (!yasayan.includes(ben.id) || !yetkili) return json({hata: 'oy veremezsin'}, 403);
        await env.DB.prepare('UPDATE oyuncular SET oy = ? WHERE id = ?')
          .bind(kirp(g.hedef, 64), ben.id).run();
        let oy = await oyuncuListe(env, kod);
        const oylar = oylariTopla(oy);
        const gerekli = o.faz === 'gece'
          ? yasayan.filter(x => o.roller[x] === 'vampir')
          : yasayan;
        if (gerekli.every(x => oylar[x])) {
          const faz = o.faz, tur = o.tur;
          if (cozumle(d, oy) && await odaYazKosullu(env, kod, d, faz, tur)) {
            await oylariSil(env, kod);
            oy.forEach(x => x.oy = null);
          } else {
            d = await odaOku(env, kod);
            oy = await oyuncuListe(env, kod);
          }
        }
        return json(vampirGorunum(d, oy, ben.id));

      } else if (p[2] === 'vdurum') {
        if (!d.oyun) return json({oyunVar: false});
        let oy = await oyuncuListe(env, kod);
        if (d.oyun.faz === 'gunduz' && d.oyun.bitis && Date.now() > d.oyun.bitis) {
          const faz = d.oyun.faz, tur = d.oyun.tur;
          if (cozumle(d, oy) && await odaYazKosullu(env, kod, d, faz, tur)) {
            await oylariSil(env, kod);
            oy.forEach(x => x.oy = null);
          } else {
            d = await odaOku(env, kod);
            oy = await oyuncuListe(env, kod);
          }
        }
        return json(vampirGorunum(d, oy, ben.id));

      } else if (p[2] === 'puan') {
        const oy = await oyuncuListe(env, kod);
        const hedef = oy[Number(g.sira)];
        if (hedef) {
          await env.DB.prepare('UPDATE oyuncular SET puan = max(0, puan + ?) WHERE id = ?')
            .bind(Number(g.delta) || 0, hedef.id).run();
        }
      } else {
        return json({hata: 'yok'}, 404);
      }
      return json(await gorunum(env, kod, d, ben.id));
    } catch (e) {
      return json({hata: String((e && e.message) || e)}, 500);
    }
  }
};
